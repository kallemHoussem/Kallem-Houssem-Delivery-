rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ⚠️ Remplace par TON email (en minuscules), celui de la gestion
    function owner() {
      return request.auth != null && request.auth.token.email == 'REMPLACE@TON-EMAIL.COM';
    }

    // Commandes : tout le monde peut en créer une (avec ou sans compte).
    // Lecture : toi, ou le client qui l'a créée.
    match /orders/{id} {
      allow create: if request.resource.data.keys().hasOnly(['kind','nom','adr','text','total','status','ts','tel','maps','uid'])
                    && request.resource.data.status == 'new'
                    && request.resource.data.text is string
                    && request.resource.data.text.size() < 3000
                    && request.resource.data.ts is timestamp
                    && (!('uid' in request.resource.data)
                        || (request.auth != null && request.resource.data.uid == request.auth.uid));
      allow read: if owner() || (request.auth != null && resource.data.uid == request.auth.uid);
      allow update, delete: if owner();
    }

    // Profil client : chaque client voit et modifie seulement le sien. Toi, tu peux lire.
    match /clients/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
      allow read: if owner();
    }

    // Gestion (livraisons, carburant, charges) : seulement toi
    match /gestion/{doc} {
      allow read, write: if owner();
    }
  }
}
